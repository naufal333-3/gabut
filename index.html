function hitungBMI(){ let umur = parseInt(document.getElementById("umur").value); let berat = parseFloat(document.getElementById("berat").value); let tinggi = parseFloat(document.getElementById("tinggi").value) / 100; if(!umur || !berat || !tinggi){ alert("Lengkapi semua data!"); return; } let bmi = berat / (tinggi * tinggi); let kategori = ""; let warna = ""; let persen = Math.min((bmi / 40) * 100, 100); if(bmi < 18.5){ kategori = "Kurus"; warna = "#3B82F6"; // Biru } else if(bmi < 25){ kategori = "Normal"; warna = "#22C55E"; // Hijau } else if(bmi < 30){ kategori = "Overweight"; warna = "#FACC15"; // Kuning } else{ kategori = "Obesitas"; warna = "#EF4444"; // Merah } let bbIdealMin = (18.5 * (tinggi * tinggi)).toFixed(1); let bbIdealMax = (24.9 * (tinggi * tinggi)).toFixed(1); let kalori = Math.round(berat * 30); let saran = ""; if(umur < 18){ saran = "Usia remaja memerlukan pola makan dan aktivitas yang seimbang untuk pertumbuhan."; } else if(umur <= 40){ saran = "Pertahankan pola hidup sehat dan olahraga rutin."; } else if(umur <= 60){ saran = "Jaga berat badan ideal untuk mengurangi risiko penyakit kronis."; } else{ saran = "Perhatikan massa otot, kesehatan tulang, dan konsumsi nutrisi yang cukup."; } document.getElementById("result").style.display = "block"; document.getElementById("bmiText").innerHTML = `BMI Kamu: ${bmi.toFixed(1)}`; document.getElementById("kategori").innerHTML = `Kategori: ${kategori}`; document.getElementById("kategori").style.background = warna; document.getElementById("kategori").style.color = "white"; document.getElementById("bar").style.width = persen + "%"; document.getElementById("bar").style.background = warna; document.getElementById("ideal").innerHTML = `Berat badan ideal: ${bbIdealMin} - ${bbIdealMax} kg`; document.getElementById("kalori").innerHTML = `Estimasi kebutuhan kalori: ${kalori} kkal/hari`; document.getElementById("saran").innerHTML = `Saran: ${saran}`; document.querySelector(".container").style.border = "3px solid " + warna; }
